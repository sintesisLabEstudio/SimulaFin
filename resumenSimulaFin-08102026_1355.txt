FLUJO COMPLETO DE APLICACION:
    1. INICIO Y AUTENTICACION
        El flujo arranca oficialmente en LoginActivity (configurada como LAUNCHER en AndroidManifest.xml).
        En onCreate(), llama al método generaBaseDatos() de BaseDatosHelper, que comprueba si existe 'simulafin.db' y crea las 5 tablas si no existen.
        El usuario ingresa su usuario y contraseña en etUsuario y etPassword. Al presionar btnIngresar, se valida el acceso y se redirige con Intent a MainActivity llamando a finish().

    2. MENU PRINCIPAL (MAIN ACTIVITY)
        MainActivity presenta 5 opciones centrales:
        - btnSimulaciones ─> Abre MenuSimActivity (Submenú de simulaciones)
        - btnCuentas ─> Abre CuentasActivity (Gestión de cuentas)
        - btnItems ─> Abre ItemsActivity (Gestión de ítems)
        - btnRegistro ─> Abre RegistroActivity (Registro de usuarios)
        - btnRespaldo ─> Abre RespaldoActivity (Copias de seguridad)

    3. SUBMENU DE SIMULACIONES (MENU SIM ACTIVITY)
        MenuSimActivity permite elegir entre dos modalidades de simulación:
        - btnPasoApaso ─> Abre InputSimCuentaActivity (Inicio de flujo por etapas)
        - btnSintesis ─> Abre SimulacionActivity (Pantalla dividida 60% arriba y 40% abajo con Fragments)
        - btnMenuPrincipal ─> Regresa a MainActivity

    4. FLUJO DE SIMULACION PASO A PASO
        - InputSimCuentaActivity (Paso 1): Selecciona cuenta y monto. El botón btnOkCuenta avanza a InputSimItemsActivity.
        - InputSimItemsActivity (Paso 2): Elige ítems y ajusta valores. El botón btnOkItems avanza a OutSimResumenActivity.
        - OutSimResumenActivity (Paso 3): Presenta el resumen completo de la simulación. Contiene los botones:
          * btnItems ("Volver a Ítems") ─> Llama a InputSimItemsActivity
          * btnSim ("Volver a Simulación Cuenta") ─> Llama a InputSimCuentaActivity
          * btnMenu ("Volver a Menú") ─> Llama a MenuSimActivity

    5. MANTENIMIENTO Y RESPALDO
        CuentasActivity e ItemsActivity gestionan los registros en las tablas 'Cuenta' e 'Item'.
        RegistroActivity contiene los botones btnResapaldar y btnRecuperar que navegan hacia RespaldoActivity.
        RespaldoActivity invoca las funciones de GestorRespaldo para copiar o importar la base local 'simulafin.db'.

package com.simulafin

import android.content.Intent
import android.os.Bundle
import android.widget.Button
import android.widget.EditText
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
import com.simulafin.clases.BaseDatosHelper

class LoginActivity : AppCompatActivity() {

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_login)

        // Verificar e inicializar la base de datos al arrancar la app
        val dbHelper = BaseDatosHelper(this)
        dbHelper.generaBaseDatos()

        val etUsuario = findViewById<EditText>(R.id.etUsuario)
        val etPassword = findViewById<EditText>(R.id.etPassword)
        val btnIngresar = findViewById<Button>(R.id.btnIngresar)

        btnIngresar.setOnClickListener {
            val usuario = etUsuario.text.toString()
            val password = etPassword.text.toString()

            if (usuario.isNotEmpty() && password.isNotEmpty()) {
                Toast.makeText(this, "Bienvenido $usuario", Toast.LENGTH_SHORT).show()

                // Navegar al menú principal (MainActivity)
                val intentMain = Intent(this, MainActivity::class.java)
                startActivity(intentMain)
                finish() // Cierra LoginActivity para que no quede en el historial de navegación
            } else {
                Toast.makeText(this, "Por favor ingrese usuario y contraseña", Toast.LENGTH_SHORT).show()
            }
        }
    }
}

        DETALLE: Administrador del esquema SQLite.

package com.simulafin.clases

import android.content.Context
import android.database.sqlite.SQLiteDatabase
import android.database.sqlite.SQLiteOpenHelper
import java.io.File

class BaseDatosHelper(private val context: Context) : SQLiteOpenHelper(context, DATABASE_NAME, null, DATABASE_VERSION) {

    // VERIFICA LA EXISTENCIA Y CREA LA BASE DE DATOS SI NO EXISTE
    fun generaBaseDatos(): Boolean {
        val dbFile: File = context.getDatabasePath(DATABASE_NAME)
        return if (!dbFile.exists()) {
            val db = this.writableDatabase
            db.close()
            true
        } else {
            false
        }
    }

    override fun onCreate(db: SQLiteDatabase?) {
        db?.execSQL(
            """
            CREATE TABLE Usuario (
                idUsuario INTEGER PRIMARY KEY AUTOINCREMENT,
                nombreUsuario TEXT NOT NULL,
                claveUsuario TEXT NOT NULL
            )
            """.trimIndent()
        )

        db?.execSQL(
            """
            CREATE TABLE Cuenta (
                idCuenta INTEGER PRIMARY KEY AUTOINCREMENT,
                idUsuario INTEGER NOT NULL,
                nombreCuenta TEXT NOT NULL,
                totalCuenta INTEGER NOT NULL,
                FOREIGN KEY (idUsuario) REFERENCES Usuario(idUsuario) ON DELETE CASCADE
            )
            """.trimIndent()
        )

        db?.execSQL(
            """
            CREATE TABLE Item (
                idItem INTEGER PRIMARY KEY AUTOINCREMENT,
                idUsuario INTEGER NOT NULL,
                nombreItem TEXT NOT NULL,
                valorItem INTEGER NOT NULL,
                FOREIGN KEY (idUsuario) REFERENCES Usuario(idUsuario) ON DELETE CASCADE
            )
            """.trimIndent()
        )

        db?.execSQL(
            """
            CREATE TABLE Simulacion (
                idSim INTEGER PRIMARY KEY AUTOINCREMENT,
                idCuenta INTEGER NOT NULL,
                nombreSim TEXT NOT NULL,
                fechaHoraSim TEXT NOT NULL,
                totalDisponible INTEGER NOT NULL,
                totalEgresos INTEGER NOT NULL,
                meses INTEGER NOT NULL,
                excedente INTEGER NOT NULL,
                FOREIGN KEY (idCuenta) REFERENCES Cuenta(idCuenta) ON DELETE CASCADE
            )
            """.trimIndent()
        )

        db?.execSQL(
            """
            CREATE TABLE DetalleSimulacion (
                idDetalle INTEGER PRIMARY KEY AUTOINCREMENT,
                idSim INTEGER NOT NULL,
                idItem INTEGER,
                nombreItem TEXT NOT NULL,
                valorItem INTEGER NOT NULL,
                FOREIGN KEY (idSim) REFERENCES Simulacion(idSim) ON DELETE CASCADE,
                FOREIGN KEY (idItem) REFERENCES Item(idItem) ON DELETE SET NULL
            )
            """.trimIndent()
        )
    }

    override fun onUpgrade(db: SQLiteDatabase?, oldVersion: Int, newVersion: Int) {
        db?.execSQL("DROP TABLE IF EXISTS DetalleSimulacion")
        db?.execSQL("DROP TABLE IF EXISTS Simulacion")
        db?.execSQL("DROP TABLE IF EXISTS Item")
        db?.execSQL("DROP TABLE IF EXISTS Cuenta")
        db?.execSQL("DROP TABLE IF EXISTS Usuario")
        onCreate(db)
    }

    companion object {
        const val DATABASE_NAME = "simulafin.db"
        private const val DATABASE_VERSION = 1
    }
}
